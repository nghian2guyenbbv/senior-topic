-1 - enable msql server agent
MSSQL_AGENT_ENABLED: "true"

0 - craeate a new database
CREATE DATABASE my_db;


1 - enable CDC for database
use my_db;
EXEC sys.sp_cdc_enable_db;

1.1 Create employee table
CREATE TABLE dbo.Employee (ID INT identity, name varchar(255), age int);

2- enable CDC for table

EXEC sys.sp_cdc_enable_table
 @source_schema = 'dbo',
 @source_name = 'Employee',
 @role_name = NULL;
 
3- Verify cdc status
select name, is_cdc_enabled
from sys.databases
where name = 'my_db';

4- check cdc if enabled for table
select * from cdc.change_tables;


SELECT  
    servicename,
    status_desc,
    startup_type_desc
FROM sys.dm_server_services
WHERE servicename LIKE 'SQL Server Agent%';
